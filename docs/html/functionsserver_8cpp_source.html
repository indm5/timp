61// ЗАДАЧА 3: Разбиение двудольного графа на доли
62QJsonObject GraphFunctions::task3_BipartiteParts(const QJsonArray &edges, int vertices)
63{
64 QJsonObject result;
65 QJsonArray partA, partB;
66
67 if (vertices == 0) {
68 result["status"] = "error";
69 result["message"] = "Граф не содержит вершин";
70 return result;
71 }
72
73 // Построение списка смежности
74 QVector<QVector<int>> adj(vertices);
75 for (const auto &edge : edges) {
76 QJsonArray edgeArray = edge.toArray();
77 if (edgeArray.size() < 2) continue;
78 int u = edgeArray[0].toInt();
79 int v = edgeArray[1].toInt();
80 if (u >= 0 && u < vertices && v >= 0 && v < vertices) {
81 adj[u].append(v);
82 adj[v].append(u);
83 }
84 }
85
86 // BFS раскраска в 2 цвета
87 QVector<int> color(vertices, -1);
88 QQueue<int> q;
89 bool isBipartite = true;
90
91 for (int i = 0; i < vertices; i++) {
92 if (color[i] == -1) {
93 q.enqueue(i);
94 color[i] = 0;
95
96 while (!q.isEmpty()) {
97 int u = q.dequeue();
98 for (int v : adj[u]) {
99 if (color[v] == -1) {
100 color[v] = color[u] ^ 1;
101 q.enqueue(v);
102 } else if (color[v] == color[u]) {
103 isBipartite = false;
104 break;
105 }
106 }
107 if (!isBipartite) break;
108 }
109 }
110 if (!isBipartite) break;
111 }
112
113 if (!isBipartite) {
114 result["status"] = "error";
115 result["action"] = "task3_result";
116 result["is_bipartite"] = false;
117 result["message"] = "Граф НЕ является двудольным! Разбиение на доли невозможно.";
118 return result;
119 }
120
121 // Формирование долей
122 for (int i = 0; i < vertices; i++) {
123 if (color[i] == 0) {
124 partA.append(i);
125 } else {
126 partB.append(i);
127 }
128 }
129
130 result["status"] = "success";
131 result["action"] = "task3_result";
132 result["is_bipartite"] = true;
133 result["partA"] = partA;
134 result["partB"] = partB;
135
136 QString aStr, bStr;
137 for (int i = 0; i < partA.size(); i++) aStr += QString::number(partA[i].toInt()) + " ";
138 for (int i = 0; i < partB.size(); i++) bStr += QString::number(partB[i].toInt()) + " ";
139 result["message"] = QString("Граф двудольный! Доля A: [%1] Доля B: [%2]").arg(aStr).arg(bStr);
